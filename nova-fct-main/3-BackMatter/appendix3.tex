\chapter{Grow-only Set}
\label{app:grow_set}

\begin{Scala}
import org.verifx.practical.crdts.CvRDT
import org.verifx.practical.crdts.CvRDTProof1

class GSet[elem](set: Set[elem] = new Set[elem]()) extends CvRDT[GSet[elem]] {

  def add(v: elem) = new GSet(this.set.add(v))

  def lookup(v: elem) = this.set.contains(v)

  def merge(that: GSet[elem]) = new GSet(this.set.union(that.set))

  def compare(that: GSet[elem]) = this.set.subsetOf(that.set)
  
}

object GSet extends CvRDTProof1[GSet]
\end{Scala}
