\chapter{Positive-Negative Counter}
\label{app:positive_negative_counter}

\begin{Scala}
import org.verifx.practical.crdts.CvRDT
import org.verifx.practical.crdts.CvRDTProof

class PNCounter_CRDT(incs: Int, decs: Int) extends CvRDT[PNCounter_CRDT] {

  def value() = {
    (this.incs - this.decs)
  }

  private def max(a: Int, b: Int) = {
    if (a >= b) a else b
  }

  def merge(that: PNCounter_CRDT) = {
    new PNCounter_CRDT(this.max(this.incs, that.incs), this.max(this.decs, that.decs))
  }

  def compare(that: PNCounter_CRDT): Boolean = {
    this.incs == that.incs && this.decs == that.decs
  }

  def increment(v: Int) = {
    new PNCounter_CRDT(this.incs + v, this.decs)
  }

  def decrement(v: Int) = {
    new PNCounter_CRDT(this.incs, this.decs + v)
  }

}

object PNCounter_CRDT extends CvRDTProof[PNCounter_CRDT]
\end{Scala}
